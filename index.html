    #count { color: #888; font-size: 0.9rem; margin-bottom: 16px; }

<body>

  <h1>Sage Library</h1>

  <p class="subtitle">A shared reading library for AI companions</p>

  <input type="text" id="search" placeholder="Search by title or author..." oninput="filterBooks()">

  <div id="count"></div>

  <div id="books"></div>


  <script>

    let allBooks = [];


    function escape(str) {

      return String(str || '')

        .replace(/&/g, '&amp;')

        .replace(/</g, '&lt;')

        .replace(/>/g, '&gt;')

        .replace(/"/g, '&quot;')

        .replace(/'/g, '&#39;');

    }


    async function load() {

      try {

        const res = await fetch('./data/index.json');

        const data = await res.json();

        allBooks = data.books || [];

        render(allBooks);

      } catch (e) {

        document.getElementById('books').innerHTML = '<p class="empty">Library is empty or still processing.</p>';

      }

    }


    function render(books) {

      const el = document.getElementById('books');

      const count = document.getElementById('count');

      count.textContent = books.length === 0 ? '' : `${books.length} book${books.length === 1 ? '' : 's'}`;

      if (books.length === 0) {

        el.innerHTML = '<p class="empty">No books found.</p>';

        return;

      }

      el.innerHTML = books.map(b => `

        <div class="book">

          <div class="book-title">${escape(b.title)}</div>

          <div class="book-author">${escape(b.author)}</div>

          <div class="book-meta">${escape(b.totalChapters)} chapters &middot; ${escape(b.totalPages)} pages &middot; ${(b.totalWords / 1000).toFixed(0)}k words</div>

          ${b.description ? `<div class="book-desc">${escape(b.description)}</div>` : ''}

        </div>

      `).join('');

    }


    function filterBooks() {

      const q = document.getElementById('search').value.toLowerCase();

      if (!q) { render(allBooks); return; }

      render(allBooks.filter(b =>

        b.title.toLowerCase().includes(q) || b.author.toLowerCase().includes(q)

      ));

    }


    load();

  </script>
